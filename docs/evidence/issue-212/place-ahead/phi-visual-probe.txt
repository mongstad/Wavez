extends "res://ui/shell/test_orbit_center_view.gd"
func run() -> void:
	output = "/tmp/wavez-212-place-ahead/visual-final"
	var view = Viewer.new(); root.add_child(view); view.set_process(false)
	view._open_capture("/tmp/wavez-212-automatic/equator/equator-field-stationary.wvz"); view.playing = false; await ready(view)
	view.orbit_enabled = false; view.camera_target = Vector3(2,1,-1); view.camera_distance = 8.; view._update_camera()
	view.shell.edit_orbit_center(); var editor = view.shell.orbit_editor
	fields(editor,Vector3(2,1,-1),.6)
	view.environment.glow_enabled = false
	view.scene_layers.states.field.opacity = 2.; view.scene_layers.resample()
	for axis in 3:
		var rod := MeshInstance3D.new(); var mesh := BoxMesh.new(); var size := Vector3.ONE*.025; size[axis] = 1.8; mesh.size = size; rod.mesh = mesh; rod.position = Vector3(2,1,-1); view.add_child(rod)
		var material := StandardMaterial3D.new(); material.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED; material.albedo_color = [Color.RED,Color.GREEN,Color.CORNFLOWER_BLUE][axis]; material.no_depth_test = true; material.render_priority = 10; rod.material_override = material
	await screenshot("placed-normal")
	var pose: Transform3D = view.camera.transform; var radius: float = editor.radius
	var depth_before: float = -(view.camera.global_transform.affine_inverse()*editor.center).z
	var size_before: float = view.camera.unproject_position(editor.center+view.camera.basis.x*radius).distance_to(view.camera.unproject_position(editor.center))
	var button: Button
	for node in editor.panel.get_child(0).get_children()[-1].get_children():
		if node is Button and node.text == "Place ahead": button = node
	var press := InputEventMouseButton.new(); press.button_index = MOUSE_BUTTON_LEFT; press.pressed = true; press.position = button.get_global_rect().get_center()
	root.push_input(press,true); press = press.duplicate(); press.pressed = false; root.push_input(press,true)
	var placed_depth: float = -(view.camera.global_transform.affine_inverse()*editor.center).z
	var placed_size: float = view.camera.unproject_position(editor.center+view.camera.basis.x*radius).distance_to(view.camera.unproject_position(editor.center))
	check(view.camera.transform == pose and editor.radius == radius,"Place ahead retains exact camera pose and radius before navigation")
	check(is_equal_approx(placed_depth,depth_before) and is_equal_approx(placed_size,size_before),"Place ahead retains placement depth and projected radius")
	await screenshot("ahead-normal")
	var fixed: Vector3 = editor.center
	focus_none(); var event := InputEventKey.new(); event.physical_keycode = KEY_A; event.pressed = true; Input.parse_input_event(event); Input.flush_buffered_events()
	view.set_process(true)
	for frame in 24: await process_frame
	view.set_process(false); event = event.duplicate(); event.pressed = false; Input.parse_input_event(event); Input.flush_buffered_events()
	check(editor.center == fixed and editor.radius == radius,"Actual WASD process loop retains exact placed XYZ and radius")
	await screenshot("inspect-after-A-normal")
	view.shell.enter_immersive(); await screenshot("inspect-fullscreen")
	var depth_after: float = -(view.camera.global_transform.affine_inverse()*editor.center).z
	var size_after: float = view.camera.unproject_position(editor.center+view.camera.basis.x*radius).distance_to(view.camera.unproject_position(editor.center))
	check(editor.radius == radius,"Place ahead leaves exact numeric radius unchanged")
	var report := {"source_commit":"candidate working tree after depth fix","checks":checks,"failures":failures,"radius":radius,"depth_before":depth_before,"depth_after_place_ahead":placed_depth,"projected_radius_before":size_before,"projected_radius_after_place_ahead":placed_size}
	var file := FileAccess.open(output.path_join("cause.json"),FileAccess.WRITE); file.store_string(JSON.stringify(report,"\t")); file.close()
	view.queue_free(); for frame in 3: await process_frame
	print(JSON.stringify(report)); quit(0 if failures.is_empty() else 1)
