extends "res://ui/shell/test_focus_navigation.gd"
class MouseState:
	extends Node
	func _input(event: InputEvent):
		if event.device == 9001: get_viewport().set_input_as_handled()
var areas: Array = []
func send_key(code: Key, window: Window, ctrl := false, shift := false, unicode_value := 0):
	var event := InputEventKey.new(); event.keycode = code; event.pressed = true
	event.ctrl_pressed = ctrl; event.shift_pressed = shift; event.unicode = unicode_value; event.window_id = window.get_window_id()
	window.window_input.emit(event)
	if window.visible: window.push_input(event)
	var release := event.duplicate(); release.pressed = false
	if window.visible: window.push_input(release)
func text_key(text: String, window: Window):
	for character in text: send_key(KEY_NONE, window, false, false, character.unicode_at(0))
func mouse(point: Vector2, window: Window, pressed: bool, button := MOUSE_BUTTON_LEFT):
	if window.get_node_or_null("MouseState") == null:
		var blocker = MouseState.new(); blocker.name = "MouseState"; window.add_child(blocker)
	var event := InputEventMouseButton.new(); event.position = point; event.global_position = point
	event.button_index = button; event.pressed = pressed; event.window_id = window.get_window_id()
	# Update global button state without dispatching the dialog-local coordinates twice.
	var state := event.duplicate(); state.device = 9001
	if window.is_embedded(): state.position += Vector2(window.position); state.global_position = state.position
	Input.parse_input_event(state); Input.flush_buffered_events()
	window.push_input(event)
func mouse_click(point: Vector2, window: Window, button := MOUSE_BUTTON_LEFT):
	mouse(point,window,true,button); await settle(); mouse(point,window,false,button); await settle()
func picture(name: String, window: Window):
	await settle(); await RenderingServer.frame_post_draw
	window.get_texture().get_image().save_png(output.path_join(name+".png"))
func reveal(editor: LineEdit):
	var parent := editor.get_parent()
	while parent != null:
		if parent is ScrollContainer: parent.ensure_control_visible(editor)
		parent = parent.get_parent()
	await settle()
func matrix(editor: LineEdit, name: String):
	var window: Window = editor.get_window()
	var original_text := editor.text
	if editor.text.is_empty(): editor.text = "123.45"
	var initial := editor.text
	if editor.get_parent() is SpinBox:
		initial = initial.trim_prefix(editor.get_parent().prefix).trim_suffix(editor.get_parent().suffix).strip_edges()
	await reveal(editor)
	var available: Array = nav.chain(window); var index := available.find(editor)
	check(index >= 0,name+": editor in existing focus order")
	if index < 0: editor.text = original_text; return
	available[posmod(index-1,available.size())].grab_focus(); await settle()
	if available.size() == 1: editor.release_focus()
	send_key(KEY_TAB,window); await settle()
	check(window.gui_get_focus_owner() == editor and editor.get_selected_text() == initial,name+": Tab selects full value")
	send_key(KEY_TAB,window); await settle();send_key(KEY_TAB,window,false,true);await settle()
	check(window.gui_get_focus_owner() == editor and editor.get_selected_text() == initial,name+": Shift+Tab selects full value")
	available[posmod(index-1,available.size())].grab_focus();await reveal(editor)
	editor.release_focus();await settle()
	await mouse_click(editor.get_global_rect().get_center(),window);await settle()
	check(window.gui_get_focus_owner() == editor and editor.get_selected_text() == initial,name+": first mouse focus selects full value")
	await picture(name,window)
	text_key("7",window);await settle()
	check(editor.text == "7",name+": first typed character replaces full value")
	editor.text = initial
	await mouse_click(editor.get_global_rect().get_center(),window);await settle()
	check(not editor.has_selection(),name+": second click positions caret")
	var column := editor.caret_column
	text_key("8",window);await settle()
	check(editor.text == initial.insert(column,"8"),name+": partial insertion at chosen caret")
	editor.text = initial;editor.select_all();send_key(KEY_RIGHT,window);await settle()
	check(not editor.has_selection(),name+": arrow releases selection")
	editor.grab_focus();nav.reconcile();await settle()
	check(not editor.has_selection(),name+": repeated focus/reconcile does not reselect")
	editor.text = original_text
	areas.append(name)
func partial_matrix(editor: LineEdit):
	var window: Window = editor.get_window();var initial := editor.text
	editor.text = "123.45";editor.grab_focus();editor.select_all()
	send_key(KEY_LEFT,window);send_key(KEY_RIGHT,window);send_key(KEY_RIGHT,window,false,true);await settle()
	check(editor.get_selected_text() == "2","Shift+Arrow selects a substring")
	text_key("9",window);await settle();check(editor.text == "193.45","Typing replaces only Shift+Arrow substring")
	editor.text = "123.45";editor.deselect();editor.caret_column = 0
	var font := editor.get_theme_font("font");var font_size := editor.get_theme_font_size("font_size")
	var start := editor.get_global_rect().position+Vector2(editor.get_theme_stylebox("normal").content_margin_left+font.get_string_size("1",HORIZONTAL_ALIGNMENT_LEFT,-1,font_size).x,editor.size.y*.5)
	var finish := start+Vector2(font.get_string_size("23",HORIZONTAL_ALIGNMENT_LEFT,-1,font_size).x,0)
	mouse(start,window,true); await settle()
	var motion := InputEventMouseMotion.new();motion.position = finish;motion.global_position = finish;motion.relative = finish-start;motion.button_mask = MOUSE_BUTTON_MASK_LEFT;window.push_input(motion); await settle()
	mouse(finish,window,false);await settle()
	check(editor.get_selected_text() == "23","Mouse drag selects substring")
	text_key("8",window);await settle();check(editor.text == "18.45","Typing replaces only dragged substring")
	send_key(KEY_A,window,true);await settle();check(editor.get_selected_text() == editor.text,"Explicit Ctrl+A still selects all")
	var clipboard := DisplayServer.clipboard_get()
	send_key(KEY_C,window,true);await settle();check(DisplayServer.clipboard_get() == "18.45","Ctrl+C copies selection")
	DisplayServer.clipboard_set("321.75");send_key(KEY_V,window,true);await settle();check(editor.text == "321.75","Ctrl+V replaces selection")
	DisplayServer.clipboard_set(clipboard);editor.text = initial
